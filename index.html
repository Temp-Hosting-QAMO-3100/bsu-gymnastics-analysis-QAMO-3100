    value_name="Score"

# Routines performed by gymnasts absent from 2025

    long[long["Gymnast"].isin(departed)]

    df_2025["Team"].isin(["Boise State", opponent])
].copy()


df = df.melt(
    id_vars="Team",
    value_vars=events,
    var_name="Event",
    value_name="Score"
).dropna(subset=["Score"])




scores = {}


teams_of_interest = ["Boise State", opponent]
filtered_df = df[df["Team"].isin(teams_of_interest)]


scores = filtered_df.groupby(["Event", "Team"])["Score"].apply(lambda x: x.to_numpy()).to_dict()


summary = (
    df.groupby(["Event", "Team"])["Score"]
    .agg(
        N="count",
        Mean="mean",
        Median="median",
        SD="std",
        Min="min",
        Max="max"
    )
)


# Below 9.60
summary["Below_9.60"] = (
    df.groupby(["Event", "Team"])["Score"]
    .apply(lambda x: (x < 9.60).sum())
)


summary["Below_9.60_%"] = (
    df.groupby(["Event", "Team"])["Score"]
    .apply(lambda x: 100 * (x < 9.60).mean())
)


# 9.80+
summary["9.80+"] = (
    df.groupby(["Event", "Team"])["Score"]
    .apply(lambda x: (x >= 9.80).sum())
)


# 9.90+
summary["9.90+"] = (
    df.groupby(["Event", "Team"])["Score"]
    .apply(lambda x: (x >= 9.90).sum())
)


print("\n" + "=" * 80)
print(f"BOISE STATE vs {opponent.upper()} — 2025 RAW SCORE DISTRIBUTIONS")
print("=" * 80)


print(summary.round(3))
